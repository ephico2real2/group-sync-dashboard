# binding, created before the walk and deleted after it. Commands run with the lab's scratch kubeconfig

$ oc get clusterrolebindings -o json | jq -r '.items[] | select(any(.subjects[]?; .kind=="User" and .name=="developer")) | .metadata.name'
(no output: developer held no cluster role before the walk)

$ oc create clusterrolebinding walk-322-developer-cluster-reader --clusterrole=cluster-reader --user=developer
$ oc label clusterrolebinding walk-322-developer-cluster-reader rbac.ocp.io/config-source=walk-322
$ oc auth can-i list clusterrolebindings --as developer
yes
$ oc auth can-i update clusterrolebindings --as developer
no
